          .baud_rate  = UART_BAUD,
          .data_bits  = UART_DATA_8_BITS,
          .parity     = UART_PARITY_DISABLE,
          .stop_bits  = UART_STOP_BITS_1,
          .flow_ctrl  = UART_HW_FLOWCTRL_DISABLE,
      };

      uart_set_pin(UART_PORT, UART_TX_PIN, UART_RX_PIN, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);

  void uart_rx_task(void *pvParameter)
  {
      uint8_t buf[UART_BUF_SIZE];

      while (1) {
          int len = uart_read_bytes(UART_PORT, buf, sizeof(buf), 20 / portTICK_PERIOD_MS);
          if (len > 0) {
              // 파싱 후 큐에 전달
              int16_t value = parse_modbus_response(buf, len);
              xQueueSend(modbus_chart_queue, &value, 0);
          }
      }
  }

  Modbus RTU면 한 단계 더
  
  Modbus는 단순 스트림이 아니라 요청/응답 구조라서:

  [Master가 요청 송신] → [Slave가 응답 수신] → [파싱] → [큐에 push]

  - Master(ESP32가 질의하는 쪽): uart_write_bytes()로 Modbus 요청 프레임 송신 → uart_read_bytes()로 응답
  대기
  - Slave(ESP32가 데이터 받는 쪽): 그냥 uart_read_bytes()로 읽고 프레임 파싱

  RS485 쓰면 DE/RE 핀 제어도 필요해.

  ---
  왜 FreeRTOS를 쓰냐?
  
  ESP32는 두 가지 일을 동시에 해야 해.

  1. 화면 그리기 (LVGL — 계속 루프 돌면서 터치 처리, 화면 갱신)
  2. UART로 Modbus 통신 (요청 보내고 응답 기다리고 파싱)

  근데 이 두 가지를 하나의 루프에서 순서대로 하면 문제가 생겨.

  // 이렇게 하면 안 됨
  while(1) {
      lv_task_handler();        // 화면 처리
      uart_write_bytes(...);    // 요청 전송
      uart_read_bytes(...);     // 응답 대기 ← 여기서 300ms 멈춤!!!
      lv_task_handler();        // 300ms 뒤에야 다시 화면 처리
  }

  uart_read_bytes()가 응답 올 때까지 300ms 동안 멈춰버리면 그 사이에 화면이 끊겨. 터치도 안 먹히고.

  FreeRTOS는 이걸 해결해줌. 태스크라는 단위로 쪼개서 번갈아가며 실행시켜줘.

  Core 0: [modbus_master_task] 요청/응답/파싱만 담당
  Core 1: [lv_task_handler]   화면/터치만 담당

  ESP32는 코어가 2개라서 진짜로 동시에 실행돼.

  ---
  큐(Queue)가 뭔데?
  
  두 태스크가 데이터를 주고받는 우체통이야.

  - s_req_queue : UI → Modbus 태스크로 "이거 읽어와" 요청 전달
  - modbus_chart_queue : Modbus 태스크 → UI로 "읽어온 값" 전달

  왜 변수를 그냥 공유하면 안 되냐? 두 태스크가 동시에 같은 변수를 건드리면 데이터가 깨질 수 있어.
  큐는 이걸 내부적으로 안전하게 처리해줌.

  파일이 두 개인 이유

  ┌────────────────────────┬──────────────────────────────────┐
  │          파일          │               역할               │
  ├────────────────────────┼──────────────────────────────────┤
  │ lv_modbus_connection.c │ 통신만 담당 (UART, 프레임, 파싱) │
  ├────────────────────────┼──────────────────────────────────┤
  │ lv_modbus.c            │ UI만 담당 (차트, 위젯)           │
  └────────────────────────┴──────────────────────────────────┘

  이렇게 나눠두면 나중에 "Modbus 통신 방식 바꿔야겠다" 할 때 lv_modbus.c 안 건드리고
  lv_modbus_connection.c만 수정하면 돼. 반대도 마찬가지고.
